Table pedido {
  id integer [primary key]
}

Table itempedido {
  id integer [primary key]
  pedido_id integer
  pizza_id integer
  refrigerante_id integer
}

Table pizza {
  id integer [primary key]
  tamanho varchar
}

Table ingrediente {
  id integer [primary key]
  descricao varchar
  quantidade varchar
  medida varchar
}

Table ingredientes_pizza {
  pizza_id integer
  ingrediente_id integer
}
